class Solution {
  public:
    string encode(vector<string>& arr) {
        string res = "";
        for (string& s : arr) {
            res += to_string(s.size()) + "/:" + s;
        }
        return res;
    }

    vector<string> decode(string& s) {
        vector<string> res;
        int i = 0, n = s.size();

        while (i < n) {
            int len = 0;
                    
            while (s[i] != '/') {
                len = len * 10 + (s[i] - '0');
                i++;
            }
            
            i += 2;
            res.push_back(s.substr(i, len));
            i += len;
        }
        return res;
    }
};
